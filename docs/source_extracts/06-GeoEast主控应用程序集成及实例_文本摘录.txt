#include "ggiTools.h"
#include "baerrorrecord.h"
#include "RdApplication.h"
#include "GEDO_TypeDef.h"
int main( int argc, char **argv )
{
RdApplication app(argc, argv);
FW_ArgParser myArgs = app.args();
const int paraCount = myArgs.m_dataIndexes.count();
QString dbName = myArgs.m_db;
QString projectName, surveyName, lineName, dataName;
if( paraCount > 0 )
projectName = myArgs.m_dataIndexes.at(0)->name();
if( paraCount > 1 )
surveyName = myArgs.m_dataIndexes.at(1)->name();
if( paraCount > 2 )
lineName = myArgs.m_dataIndexes.at(2)->name();
if( paraCount > 3 && myArgs.m_dataIndexes.at(3)->id() == tyGEDOPsSeismicDataSet)
dataName = myArgs.m_dataIndexes.at(3)->name();
msgBox( dbName, projectName, surveyName, lineName, dataName );
return app.exec();
}
